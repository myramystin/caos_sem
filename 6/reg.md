## регистры 

```
; Общего назначения (64-битные)
rax, rbx, rcx, rdx  ; A, B, C, D регистры
rsi, rdi            ; Source/Destination Index
rsp, rbp            ; Stack/Base Pointer
r8, r9, r10, r11, r12, r13, r14, r15  ; Дополнительные

; 32-битные версии
eax, ebx, ecx, edx, esi, edi, esp, ebp
r8d, r9d, r10d, r11d, r12d, r13d, r14d, r15d

; 16-битные и 8-битные
ax, bx, cx, dx
al, ah, bl, bh, cl, ch, dl, dh
```
